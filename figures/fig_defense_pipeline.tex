% fig_defense_pipeline.tex
% Where each AP-side soft patch acts on the BFI sounding chain.
% Single-column. Compact, slide-deck idiom. Numeric tags mark the
% intervention points; the caption identifies which one reaches the
% chance line.
%
% Pipeline: AP --NDP--> Client; client SVDs the channel, averages
% across subcarrier groups, Givens-decomposes, quantises, sends BFI.
% Eve listens to BFI from radio range.
%
% Defenses:
%   (1) Unitary obfusc.   acts at AP TX (modifies the sounding signal)
%   (2) Subcarrier group. acts on V before Givens (averaging)
%   (3) Sound-rate cap    throttles NDP rate (clock on the NDP arrow)
%   (4) DP-SQ             acts at the quantizer (per-coordinate flip)
%
% Requires (already loaded in paper.tex):
%   \usepackage{tikz}
%   \usetikzlibrary{arrows.meta,positioning,calc,fit,backgrounds,%
%                    shapes.geometric,decorations.pathmorphing}
%
\begin{tikzpicture}[
    x=1mm, y=1mm,
    font=\sffamily\footnotesize,
    >={Stealth[length=1.6mm,width=1.3mm]},
    every node/.style={font=\sffamily\footnotesize},
]

\definecolor{cBlue}{HTML}{1F4E8C}
\definecolor{cOrange}{HTML}{C5781E}
\definecolor{cRed}{HTML}{C53E3E}
\definecolor{cGreen}{HTML}{1B7A5C}
\definecolor{cGray}{HTML}{555555}
\definecolor{cMidGray}{HTML}{888888}
\definecolor{cBlueFill}{HTML}{E0EAF5}
\definecolor{cOrangeFill}{HTML}{F8E8D0}
\definecolor{cGreenFill}{HTML}{D8ECDF}

% Canvas ~ 90 x 56 mm.
% Layout: row 0 (y=44..52) numbered defense labels, row 1 (y=30..40)
% AP/client/Eve, row 2 (y=14..24) client-side processing chain,
% row 3 (y=0..10) bottom-row labels for defenses 2 and 4.

% ---- Defense labels at the top (numbered chips + words) --------------
\node[circle, draw=cGreen, fill=cGreenFill, line width=0.5pt,
      inner sep=0pt, minimum size=3.6mm,
      font=\sffamily\scriptsize\bfseries, text=cGreen] (d1) at (8, 48) {1};
\node[anchor=west, font=\sffamily\scriptsize, color=cGreen]
  at (10.5, 48) {unitary obfusc.};

\node[circle, draw=cOrange, fill=cOrangeFill, line width=0.5pt,
      inner sep=0pt, minimum size=3.6mm,
      font=\sffamily\scriptsize\bfseries, text=cOrange] (d3) at (52, 48) {3};
\node[anchor=west, font=\sffamily\scriptsize, color=cOrange]
  at (54.5, 48) {sound-rate cap};

% ---- Endpoints (row 1) -----------------------------------------------
\node[draw=cBlue, fill=cBlueFill, line width=0.7pt, rounded corners=1.0mm,
      minimum width=10mm, minimum height=8mm, align=center, text=cBlue,
      font=\sffamily\scriptsize\bfseries]
  (ap) at (8, 35) {AP};

\node[draw=cBlue, fill=cBlueFill, line width=0.7pt, rounded corners=1.0mm,
      minimum width=10mm, minimum height=8mm, align=center, text=cBlue,
      font=\sffamily\scriptsize\bfseries]
  (cli) at (44, 35) {client};

\node[draw=cRed, fill=white, line width=0.7pt, rounded corners=1.0mm,
      minimum width=10mm, minimum height=8mm, align=center, text=cRed,
      font=\sffamily\scriptsize\bfseries]
  (eve) at (82, 35) {Eve};

% NDP arrow (AP -> client)
\draw[->, cBlue, line width=0.7pt] (ap.east) -- (cli.west)
    node[midway, above, font=\sffamily\scriptsize, color=cBlue] {NDP};

% ---- Connectors from defense (1) and (3) labels down to where they act
\draw[cGreen, line width=0.4pt] (d1.south) -- (ap.north);
\draw[cOrange, line width=0.4pt] (d3.south) -- (26, 40);
% small marker on the NDP arrow itself for defense 3
\node[circle, fill=cOrange, inner sep=0pt, minimum size=1.4mm]
  at (26, 39) {};

% ---- Client-side processing chain (row 2) ----------------------------
\node[draw=cGray, fill=white, line width=0.4pt, rounded corners=0.7mm,
      minimum width=7mm, minimum height=4mm, align=center,
      font=\sffamily\scriptsize, text=cGray]
  (svd) at (44, 20) {SVD};

\node[draw=cGray, fill=white, line width=0.4pt, rounded corners=0.7mm,
      minimum width=7mm, minimum height=4mm, align=center,
      font=\sffamily\scriptsize, text=cGray]
  (grp) at (54, 20) {avg};

\node[draw=cGray, fill=white, line width=0.4pt, rounded corners=0.7mm,
      minimum width=8mm, minimum height=4mm, align=center,
      font=\sffamily\scriptsize, text=cGray]
  (giv) at (65, 20) {Givens};

\node[draw=cGray, fill=white, line width=0.4pt, rounded corners=0.7mm,
      minimum width=7mm, minimum height=4mm, align=center,
      font=\sffamily\scriptsize, text=cGray]
  (qnt) at (76, 20) {quant};

\draw[->, cGray, line width=0.4pt] (svd.east) -- (grp.west);
\draw[->, cGray, line width=0.4pt] (grp.east) -- (giv.west);
\draw[->, cGray, line width=0.4pt] (giv.east) -- (qnt.west);

% Client to SVD
\draw[->, cGray, line width=0.4pt] (cli.south) -- (svd.north);
% Quant -> BFI -> Eve (curve up to Eve)
\draw[->, cRed, line width=0.5pt, dash pattern=on 0.6mm off 0.4mm]
    (qnt.east) -- ++(3,0)
    node[anchor=west, font=\sffamily\scriptsize\itshape, color=cRed]
      {BFI};
\draw[->, cRed, line width=0.5pt, dash pattern=on 0.6mm off 0.4mm]
    (84, 20) -- (84, 30);

% ---- Bottom-row defense markers (2) and (4) --------------------------
\node[circle, draw=cOrange, fill=cOrangeFill, line width=0.5pt,
      inner sep=0pt, minimum size=3.6mm,
      font=\sffamily\scriptsize\bfseries, text=cOrange]
  (d2) at (54, 9) {2};
\draw[cOrange, line width=0.4pt] (d2.north) -- (grp.south);
\node[anchor=north, font=\sffamily\scriptsize, color=cOrange,
      align=center]
  at (54, 7) {subcarrier\\group.};

\node[circle, draw=cOrange, fill=cOrangeFill, line width=0.5pt,
      inner sep=0pt, minimum size=3.6mm,
      font=\sffamily\scriptsize\bfseries, text=cOrange]
  (d4) at (76, 9) {4};
\draw[cOrange, line width=0.4pt] (d4.north) -- (qnt.south);
\node[anchor=north, font=\sffamily\scriptsize, color=cOrange]
  at (76, 7) {DP-SQ};

\end{tikzpicture}
