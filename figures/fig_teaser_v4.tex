% Figure 1 (v4): numbered workflow, full text width (7.0 in) by at most 1.7 in.
% Four steps read left to right, each with a circled number and a title, and
% an arrow between every pair of consecutive steps:
%   1 the TV (client, green) sends its channel report in plaintext (orange frame);
%   2 the frame fans out to three readers: gateway (provider, blue), house
%     (household, green), mast beyond a thin wall (outsider, vermilion), which
%     only listens and is the reader the next two steps follow;
%   3 the outsider keeps one labeled session, "TV, day 1";
%   4 and identifies the TV on days 2 to 13 (drawn check mark).
% Beneath the outsider and steps 3-4 a grey contrast row says what prior attacks need: many
% labeled recordings of the target (frame stack, "target in training", cross).
% Labels are 9 pt in the document font; colours are Okabe-Ito. The bounding
% box is fixed at 17.78 cm x 4.18 cm unless the preview defines \teasernobbox.

% frame glyph centred at (#1,#2) with scale #3: a document with a folded corner and three lines
\newcommand{\reportframe}[3]{%
  \begin{scope}[shift={(#1,#2)}, scale=#3]
    \draw[frame] (-0.27,-0.33) -- (0.13,-0.33) -- (0.27,-0.19) -- (0.27,0.33) -- (-0.27,0.33) -- cycle;
    \draw[frameline] (0.13,-0.33) -- (0.13,-0.19) -- (0.27,-0.19);

  \end{scope}}
\newcommand{\greyframe}[3]{%
  \begin{scope}[shift={(#1,#2)}, scale=#3]
    \draw[gframe] (-0.27,-0.33) -- (0.13,-0.33) -- (0.27,-0.19) -- (0.27,0.33) -- (-0.27,0.33) -- cycle;
    \draw[gframeline] (0.13,-0.33) -- (0.13,-0.19) -- (0.27,-0.19);
    \draw[gframeline] (-0.17,-0.1) -- (0.17,-0.1) (-0.17,0.03) -- (0.17,0.03);
  \end{scope}}
% a strip of five small frames, the first centred at (#1,#2)
\newcommand{\framestrip}[2]{%
  \foreach \i in {0,...,4} {\reportframe{#1+0.38*\i}{#2}{0.55}}}
