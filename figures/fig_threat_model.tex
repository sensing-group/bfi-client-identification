% Redesign rationale:
%   v7 was a vertical 4-stage pipeline forced into a wrapfigure at
%   0.40\columnwidth, which crushed the annotation text to unreadable
%   size. v8 is a horizontal 4-stage pipeline rendered at \columnwidth
%   inside a regular figure[t] float; each stage carries a short label
%   underneath rather than a verbose annotation box on the side. Stage 3
%   is highlighted because that is where the leak the paper attacks is.
%
% Required preamble (already in paper.tex):
%   \usepackage{tikz}
%   \usetikzlibrary{arrows.meta,positioning,calc,fit,backgrounds,
%                    shapes.geometric,decorations.pathmorphing}

\begin{tikzpicture}[
    x=1mm, y=1mm,
    font=\sffamily\footnotesize,
    >={Stealth[length=2.4mm,width=1.8mm]},
    every node/.style={font=\sffamily\footnotesize},
]

% Palette ------------------------------------------------------------------
\definecolor{s1Fill}{HTML}{E0EAF5}\definecolor{s1Edge}{HTML}{1F4E8C}
\definecolor{s2Fill}{HTML}{DDEEEA}\definecolor{s2Edge}{HTML}{009E73}

\tikzset{
  stagebox/.style={
    rectangle, rounded corners=1.6mm, line width=0.8pt,
    minimum width=27mm, minimum height=9mm,

    text=cSubText, font=\sffamily\scriptsize, align=center,

  },
  flowarrow/.style={->, draw=cGray, line width=0.8pt},
  arrowlabel/.style={font=\sffamily\scriptsize\itshape, text=cGray,

  leakflag/.style={text=cLeakTag, font=\sffamily\bfseries\scriptsize,
                   align=center},
}

% Canvas: 4 stages along x, centred vertically at y=12 ---------------------
